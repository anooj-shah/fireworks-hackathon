"""Score Claude on data/ext_polls2026.jsonl with the SAME prompt + grader as eval/run.py (eval/tasks/simbench).
Config mirrors results/simbench/claude_effort_probe_dev15*.jsonl: adaptive thinking, explicit effort, no temperature
(rejected by 5.5 models), max_tokens > 400. Usage: python run_claude.py <model> <effort> <out.jsonl>"""
import json, sys, time
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
REPO = Path(".")  # repo root (local absolute path removed before commit)
sys.path.insert(0, str(REPO / "eval"))
from dotenv import load_dotenv
load_dotenv(REPO / ".env")
import anthropic
from tasks import simbench as task
model, effort, out = sys.argv[1], sys.argv[2], sys.argv[3]
items = [json.loads(l) for l in open(REPO / "data/ext_polls2026.jsonl") if l.strip()]
client = anthropic.Anthropic()
def one(i, item):
    msgs = task.build_messages(item)
    sysmsg = "\n".join(m["content"] for m in msgs if m["role"] == "system")
    try:
        t0 = time.perf_counter()
        r = client.messages.create(model=model, max_tokens=4096, system=sysmsg,
                                   messages=[m for m in msgs if m["role"] != "system"],
                                   thinking={"type": "adaptive"}, output_config={"effort": effort})
        lat = time.perf_counter() - t0
        text = "".join(b.text for b in r.content if b.type == "text")
        g = task.grade(item, text)
        return {"i": i, "id": item["id"], "group": item["group"], "ok": True, "output": text, "termination": r.stop_reason,
                "latency": lat, "ttft": None, "prompt_tokens": r.usage.input_tokens, "completion_tokens": r.usage.output_tokens,
                "model": model, "effort": effort, **g}
    except Exception as e:
        return {"i": i, "id": item["id"], "group": item["group"], "ok": False, "error": f"{type(e).__name__}: {e}"[:500], "score": 0.0}
with ThreadPoolExecutor(8) as ex:
    recs = list(ex.map(lambda x: one(*x), enumerate(items)))
with open(out, "w") as f:
    for r in recs:
        f.write(json.dumps(r) + "\n")
ok = [r for r in recs if r["ok"]]
pin, pout = {"claude-opus-5-5": (4, 20), "claude-sonnet-5-5": (2, 10), "claude-haiku-5-5": (0.1, 0.5)}[model]
cost = sum(r["prompt_tokens"] * pin + r["completion_tokens"] * pout for r in ok) / 1e6
print(json.dumps({"n": len(recs), "errors": len(recs) - len(ok), "cost_usd": round(cost, 4),
                  "m_simbench": sum(r["simbench"] for r in ok) / max(1, len(ok)),
                  "m_top1": sum(r["top1"] for r in ok) / max(1, len(ok)), "m_parse_ok": sum(r["parse_ok"] for r in ok) / max(1, len(ok))}))
